# 1. Install new dependencies

npm install bcryptjs jsonwebtoken cookie-parser google-auth-library express-rate-limit nodemailer

npm install --save-dev @types/bcryptjs @types/jsonwebtoken @types/cookie-parser @types/nodemailer


# 2. Add these to your .env file (create one if you don't have it)

# --- JWT secrets (use long random strings in production, e.g. `openssl rand -hex 32`) ---
JWT_ACCESS_SECRET=replace_with_a_long_random_string
JWT_REFRESH_SECRET=replace_with_a_different_long_random_string

# --- CORS: must match your frontend's exact origin (no wildcard, since credentials:true) ---
CORS_ORIGIN=http://localhost:5173

# --- Google OAuth ---
# Get this from Google Cloud Console > APIs & Services > Credentials > OAuth 2.0 Client IDs
# (Web application type). Same client ID is used on the frontend for the Google button.
GOOGLE_CLIENT_ID=your_google_client_id.apps.googleusercontent.com

# --- SMTP (optional - leave blank to just log codes to console) ---
SMTP_HOST=
SMTP_PORT=587
SMTP_USER=
SMTP_PASS=
SMTP_FROM="img-optimizer <noreply@yourdomain.com>"
